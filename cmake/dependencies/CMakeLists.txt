include("${PROJECT_SOURCE_DIR}/cmake/CPM.cmake")

add_subdirectory(yyjson)
add_subdirectory(cot)
add_subdirectory(vmf)
if(COT_VMF_ENABLE_WEB)
	add_subdirectory(civetweb)
endif()

file(MAKE_DIRECTORY "${PROJECT_BINARY_DIR}/licenses")
foreach(dependency cot vmf yyjson expat civetweb-c-library)
	if(NOT TARGET ${dependency})
		continue()
	endif()
	get_target_property(source ${dependency} SOURCE_DIR)
	get_filename_component(parent "${source}" DIRECTORY)
	foreach(root "${source}" "${parent}")
		foreach(notice LICENSE LICENSE.md LICENSE.txt COPYING COPYING.txt)
			if(EXISTS "${root}/${notice}")
				file(COPY "${root}/${notice}" DESTINATION
					"${PROJECT_BINARY_DIR}/licenses/${dependency}")
			endif()
		endforeach()
	endforeach()
endforeach()

if(BUILD_TESTING AND (BUILD_UNIT_TESTS OR BUILD_ACCEPTANCE_TESTS))
	add_subdirectory(googletest EXCLUDE_FROM_ALL)
	if(BUILD_ACCEPTANCE_TESTS)
		add_subdirectory(nlohmann-json EXCLUDE_FROM_ALL)
		add_subdirectory(asio EXCLUDE_FROM_ALL)
		add_subdirectory(tclap EXCLUDE_FROM_ALL)
		add_subdirectory(cucumber-cpp EXCLUDE_FROM_ALL)
	endif()
endif()
